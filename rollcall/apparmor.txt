#include <tunables/global>

# AppArmor profile for Rollcall. The Supervisor renames the profile to the
# add-on's slug when it loads it. It lets Rollcall read its own image, run the
# few programs it needs (the start script, Python and ping), use the network,
# and write only to its data folder and temporary/run-time folders.
profile ping_monitor flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Network: ping (unprivileged ICMP), port and web checks, the Home Assistant
  # API and websocket, the dashboard (8099) and agent reports (8100).
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  network netlink raw,
  network unix,

  # Ordinary file-ownership capabilities only; no NET_RAW, NET_ADMIN or SYS_*.
  capability chown,
  capability dac_override,
  capability fowner,
  capability fsetid,
  capability setgid,
  capability setuid,
  capability kill,

  # Stop/restart signals from the Supervisor, and Python stopping its own children.
  signal,

  # Read the image (Python, its libraries, the app, time zones, certificates).
  /** r,
  /{usr/,}lib/** mr,
  /usr/local/lib/** mr,

  # Programs it runs.
  /run.sh rix,
  /init ix,
  /bin/** ix,
  /sbin/** ix,
  /usr/bin/** ix,
  /usr/sbin/** ix,
  /usr/lib/bashio/** ix,
  /command/** ix,
  /package/** ix,

  # Where it writes.
  /data/ r,
  /data/** rwk,
  /tmp/ r,
  /tmp/** rwk,
  /run/** rwk,
  /var/run/** rwk,
  /dev/null rw,
  /dev/tty rw,
  /dev/pts/* rw,
  @{PROC}/@{pid}/fd/ r,
  @{PROC}/@{pid}/** r,

  # Certificates for HTTPS agent reports, read only.
  /ssl/** r,

  # Nothing else on the system is writable.
  deny /proc/sys/** w,
  deny /sys/** w,
  deny /proc/sysrq-trigger rwklx,
  deny /proc/kcore rwklx,
}
